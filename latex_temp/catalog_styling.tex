% Catalog entry title  
\newcommand{\artifacttitle}[1]{%  
  \FloatBarrier%                                    % ADDED: Clear all floats first
  \vspace{16pt}%  
  {\fontsize{24pt}{28pt}\selectfont\bfseries #1\par}%  
  \vspace{4pt}%  
}  
  
% Metadata line (gray, 10pt)  
\newcommand{\metaline}[1]{%  
  {\fontsize{10pt}{13pt}\selectfont\color{metagray}#1\par}%  
}  
  
% Metadata block  
\newcommand{\metablock}[1]{%  
  \vspace{2pt}%  
  \begin{spacing}{1.15}%  
  {\fontsize{10pt}{13pt}\selectfont\color{metagray}#1}%  
  \end{spacing}%  
  \vspace{10pt}%                                    % INCREASED from 8pt
}  
  
% Catalog number marker (small, above title, number only)
\newcommand{\catnum}[1]{%
  {\fontsize{9pt}{12pt}\selectfont\color{metagray}#1\par}%
  \vspace{2pt}%
}  
  
% Separator line  
\newcommand{\artifactsep}{%  
  \FloatBarrier%                                    % ADDED: Clear floats before separator
  \vspace{14pt}%                                    % INCREASED from 10pt
  \noindent\rule{\textwidth}{0.3pt}%  
  \vspace{14pt}%                                    % INCREASED from 10pt
}  
  
% Reference text (gray, 10pt)  
\newcommand{\referencetext}[1]{%  
  \vspace{10pt}%                                    % INCREASED from 8pt
  {\fontsize{10pt}{13pt}\selectfont\color{metagray}#1\par}%  
  \vspace{6pt}%                                     % ADDED: Space after reference
}  

% NEW: Enhanced image commands to prevent overlap
% Use this for wrapped figures
\newcommand{\artifactimage}[3][r]{%
  % #1 = placement (r or l), #2 = width, #3 = filename
  \begin{wrapfigure}{#1}{#2}%
    \centering%
    \vspace{-8pt}%  % Adjust vertical position
    \includegraphics[width=\linewidth,keepaspectratio,max height=0.4\textheight]{#3}%
    \vspace{-8pt}%  % Adjust bottom space
  \end{wrapfigure}%
}

% Use this for standalone figures (not wrapped)
\newcommand{\artifactfigure}[3][0.5]{%
  % #1 = width ratio (default 0.5), #2 = filename, #3 = caption
  \begin{figure}[H]%
    \centering%
    \includegraphics[width=#1\textwidth,keepaspectratio,max height=0.5\textheight]{#2}%
    \caption{#3}%
  \end{figure}%
  \vspace{6pt}%
}

% NEW: Clear page if less than specified space remains
\newcommand{\needspace}[1]{%
  \par\penalty-100\vskip 0pt plus 1fil\penalty 100\vskip 0pt plus -1fil%
  \ifdim\pagetotal>\dimexpr\pagegoal-#1\relax%
    \clearpage%
  \fi%
}
