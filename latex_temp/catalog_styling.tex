% ============================================================  
% Custom Commands  
% ============================================================  
  
% Catalog entry title: the object name dominates, the catalogue number is a
% small ochre marker in front of it.
\newcommand{\artifacttitle}[2]{% #1 = catalogue number, #2 = name
  \FloatBarrier%
  {\fontsize{19pt}{23pt}\selectfont\bfseries
   \textcolor{accent}{#1}\hspace{0.45em}#2\par}%
}  
  
% Chapter heading is not printed: the running head already names the chapter,
% and the object titles carry the hierarchy.
  
% Metadata line (gray, 10pt)  
\newcommand{\metaline}[1]{%  
  {\fontsize{11pt}{13.5pt}\selectfont\color{metagray}#1\par}%  
}  
  
% Metadata block  
\newcommand{\metablock}[1]{%  
  \vspace{2pt}%  
  \begin{spacing}{1.15}%  
  {\fontsize{11pt}{13.5pt}\selectfont\color{metagray}#1}%  
  \end{spacing}%  
  \vspace{10pt}%                                    % INCREASED from 8pt
}  
  
% Catalog number marker (small, above title, number only)
\newcommand{\catnum}[1]{%
  {\fontsize{9pt}{12pt}\selectfont\color{metagray}#1\par}%
  \vspace{2pt}%
}  
  
% Separator line  
% parfillskip is non-zero because of KOMA's parskip=half; without resetting it
% the full-width rule overflows the column by ~12pt.
% No \FloatBarrier here: the separator is typeset inside the measured entry
% box, where a float barrier is neither needed nor legal.
% A short, light rule separates entries without competing with the chapter rule.
\newcommand{\artifactsep}{%  
  \vspace{16pt}%
  {\parfillskip=0pt\relax
   \noindent{\color{quoteborder}\rule{0.45\textwidth}{0.2pt}}\par}%
  \vspace{16pt}%
}  
  
% Reference text (gray, 10pt) — tight list, no extra paragraph spacing
\newcommand{\referencetext}[1]{%
  \par\vspace{2pt}%
  \noindent{\fontsize{10pt}{13pt}\selectfont\color{refgray}#1\par}%
  \vspace{0pt}%
}

% Image proof line below a figure
\newcommand{\imageproof}[1]{%
  \par\vspace{3pt}%
  {\fontsize{9pt}{11.5pt}\selectfont\color{refgray}#1\par}%
}  

% NEW: Enhanced image commands to prevent overlap
% Use this for wrapped figures
\newcommand{\artifactimage}[3][r]{%
  % #1 = placement (r or l), #2 = width, #3 = filename
  \begin{wrapfigure}{#1}{#2}%
    \centering%
    \vspace{-8pt}%  % Adjust vertical position
    \includegraphics[width=\linewidth,keepaspectratio,max height=0.4\textheight]{#3}%
    \vspace{-8pt}%  % Adjust bottom space
  \end{wrapfigure}%
}

% Use this for standalone figures (not wrapped)
\newcommand{\artifactfigure}[3][0.5]{%
  % #1 = width ratio (default 0.5), #2 = filename, #3 = caption
  \begin{figure}[H]%
    \centering%
    \includegraphics[width=#1\textwidth,keepaspectratio,max height=0.5\textheight]{#2}%
    \caption{#3}%
  \end{figure}%
  \vspace{6pt}%
}

\makeatletter
% Break the page unless #1 of vertical space is still available.
% Reserves the space before \markboth so the mark cannot discard it.
\newcommand{\catalogbreak}[1]{%
  \par\penalty-100
  \begingroup
    \dimen@=#1\relax
    \dimen@ii=\pagegoal
    \advance\dimen@ii by -\pagetotal
    \ifdim\dimen@ii<\dimen@
      \aftergroup\clearpage
    \fi
  \endgroup
}
\makeatother
